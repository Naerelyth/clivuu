cmake_minimum_required(VERSION 3.24)
project(Clivuu LANGUAGES NONE)

# Cargo owns compilation. CMake owns one install tree used by every package.
if(NOT CMAKE_BINARY_DIR MATCHES "^/tmp/")
  message(FATAL_ERROR "Configure out of tree under /tmp")
endif()
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/Cargo.toml" version_line REGEX "^version = \"[0-9]+\\.[0-9]+\\.[0-9]+\"")
string(REGEX REPLACE "^version = \"([^\"]+)\".*" "\\1" CLIVUU_VERSION "${version_line}")
set(CLIVUU_CARGO_TARGET_DIR "/tmp/clivuu-target" CACHE PATH "Cargo build output")
set(CLIVUU_PREBUILT_DIR "" CACHE PATH "Use an existing release CLI/desktop pair instead of building")
set(CLIVUU_PACKAGE_OUTPUT "/tmp/clivuu-dist" CACHE PATH "Package output directory")
set(CLIVUU_PACKAGE_CONTACT "$ENV{PACKAGER}" CACHE STRING "Package maintainer name and email")
set(CLIVUU_PACKAGE_RELEASE "1" CACHE STRING "Distribution package revision")
set(CLIVUU_APPIMAGETOOL "appimagetool" CACHE STRING "Official AppImage builder")
set(CLIVUU_APPIMAGE_RUNTIME "" CACHE FILEPATH "Locally supplied type-2 AppImage runtime")
foreach(output CLIVUU_CARGO_TARGET_DIR CLIVUU_PACKAGE_OUTPUT)
  cmake_path(ABSOLUTE_PATH ${output} NORMALIZE OUTPUT_VARIABLE resolved)
  if(NOT resolved MATCHES "^/tmp/")
    message(FATAL_ERROR "${output} must be under /tmp")
  endif()
endforeach()
if(CLIVUU_PREBUILT_DIR)
  set(CLIVUU_BIN_DIR "${CLIVUU_PREBUILT_DIR}")
else()
  set(CLIVUU_BIN_DIR "${CLIVUU_CARGO_TARGET_DIR}/release")
  find_program(CARGO cargo REQUIRED)
  add_custom_target(rust ALL
    COMMAND "${CARGO}" build --release --locked -p clivuu -p clivuu-desktop --bins
            --target-dir "${CLIVUU_CARGO_TARGET_DIR}"
    WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
endif()

set(CLIVUU_ARCH "${CMAKE_HOST_SYSTEM_PROCESSOR}")
if(CLIVUU_ARCH MATCHES "^(amd64|x86_64)$")
  set(CLIVUU_ARCH x86_64)
  set(CLIVUU_DEB_ARCH amd64)
  set(CLIVUU_ELF_MACHINE 3e00)
elseif(CLIVUU_ARCH MATCHES "^(arm64|aarch64)$")
  set(CLIVUU_ARCH aarch64)
  set(CLIVUU_DEB_ARCH arm64)
  set(CLIVUU_ELF_MACHINE b700)
else()
  message(FATAL_ERROR "Native packages currently support x86_64 and aarch64 hosts")
endif()
configure_file(packaging/VerifyBinaries.cmake.in VerifyBinaries.cmake @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/VerifyBinaries.cmake")
install(PROGRAMS "${CLIVUU_BIN_DIR}/clivuu" DESTINATION bin)
install(PROGRAMS "${CLIVUU_BIN_DIR}/clivuu-desktop" DESTINATION libexec)
install(FILES assets/icons/clivoo.svg assets/icons/clivoo-transparent.svg assets/icons/github-symbolic.svg DESTINATION share/clivuu/icons)
install(FILES assets/icons/github-LICENSE.txt DESTINATION share/licenses/clivuu RENAME Octicons-LICENSE.txt)
install(FILES assets/icons/clivoo.svg DESTINATION share/icons/hicolor/scalable/apps RENAME io.github.naerelyth.clivuu.svg)
install(FILES packaging/io.github.naerelyth.clivuu.desktop DESTINATION share/applications)
install(FILES README.md README.zh-CN.md LICENSE DESTINATION share/doc/clivuu)
install(DIRECTORY docs DESTINATION share/doc/clivuu FILES_MATCHING PATTERN "*.md")
install(FILES assets/icons/clivoo.svg DESTINATION share/doc/clivuu/assets/icons)

set(CPACK_PACKAGE_NAME clivuu)
set(CPACK_PACKAGE_VERSION "${CLIVUU_VERSION}")
set(CPACK_PACKAGE_VENDOR "Clivuu contributors")
set(CPACK_PACKAGE_CONTACT "${CLIVUU_PACKAGE_CONTACT}")
set(CPACK_PACKAGE_DESCRIPTION_SUMMARY "Screenshot, annotation, image pinning and region recording")
set(CPACK_RESOURCE_FILE_LICENSE "${CMAKE_CURRENT_SOURCE_DIR}/LICENSE")
set(CPACK_PACKAGE_DIRECTORY "${CLIVUU_PACKAGE_OUTPUT}")
set(CPACK_PACKAGE_FILE_NAME "clivuu-${CLIVUU_VERSION}-linux-${CLIVUU_ARCH}")
set(CPACK_PACKAGING_INSTALL_PREFIX /usr)
set(CPACK_GENERATOR TGZ)
set(CPACK_VERBATIM_VARIABLES YES)

# Debian/Ubuntu binary dependencies are derived on the target distribution.
set(CPACK_DEBIAN_FILE_NAME DEB-DEFAULT)
set(CPACK_DEBIAN_PACKAGE_RELEASE "${CLIVUU_PACKAGE_RELEASE}")
set(CPACK_DEBIAN_PACKAGE_ARCHITECTURE "${CLIVUU_DEB_ARCH}")
set(CPACK_DEBIAN_PACKAGE_SECTION graphics)
set(CPACK_DEBIAN_PACKAGE_SHLIBDEPS ON)
set(CPACK_DEBIAN_PACKAGE_DEPENDS "libgtk-4-1 (>= 4.14), libadwaita-1-0 (>= 1.5), libgstreamer1.0-0 (>= 1.20), libgtk4-layer-shell0, gstreamer1.0-pipewire, gstreamer1.0-pulseaudio, gstreamer1.0-plugins-base, gstreamer1.0-plugins-good, xdg-desktop-portal, ffmpeg")
set(CPACK_DEBIAN_PACKAGE_RECOMMENDS "xwayland, pipewire-pulse | pulseaudio")
set(CPACK_DEBIAN_PACKAGE_SUGGESTS "libtesseract5, tesseract-ocr-eng, zbar-tools")

# Fedora package names; ELF requirements remain enabled for the actual build.
set(CPACK_RPM_FILE_NAME RPM-DEFAULT)
set(CPACK_RPM_PACKAGE_RELEASE "${CLIVUU_PACKAGE_RELEASE}")
set(CPACK_RPM_PACKAGE_ARCHITECTURE "${CLIVUU_ARCH}")
set(CPACK_RPM_PACKAGE_LICENSE GPL-3.0-or-later)
set(CPACK_RPM_PACKAGE_GROUP Applications/Graphics)
set(CPACK_RPM_PACKAGE_AUTOREQ ON)
set(CPACK_RPM_PACKAGE_RELOCATABLE OFF)
set(CPACK_RPM_PACKAGE_REQUIRES "gtk4 >= 4.14, libadwaita >= 1.5, gstreamer1 >= 1.20, gtk4-layer-shell, pipewire-gstreamer, gstreamer1-plugins-base, gstreamer1-plugins-good, xdg-desktop-portal, ffmpeg-free")
set(CPACK_RPM_PACKAGE_RECOMMENDS "xorg-x11-server-Xwayland, (pipewire-pulseaudio or pulseaudio)")
set(CPACK_RPM_PACKAGE_SUGGESTS "tesseract, tesseract-langpack-eng, zbar")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/rpm-tmp")
set(CPACK_RPM_SPEC_MORE_DEFINE "%define _tmppath ${CMAKE_CURRENT_BINARY_DIR}/rpm-tmp")

configure_file(packaging/AppImage.cmake.in AppImage.cmake @ONLY)
set(CPACK_EXTERNAL_ENABLE_STAGING ON)
set(CPACK_EXTERNAL_PACKAGE_SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/AppImage.cmake")
set(CPACK_PROJECT_CONFIG_FILE "${CMAKE_CURRENT_SOURCE_DIR}/packaging/PackageOptions.cmake")
include(CPack)
